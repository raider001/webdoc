OBJ = $(SRessources:.c=.o)

.PHONY: all clean

all: app

app: $(OBJ)
	$(CC) $(CFLAGS) -o $@ $^

%.o: %.c
	$(CC) $(CFLAGS) -c $< -o $@

clean:

Shell

#!/usr/bin/env bash
set -euo pipefail

# Greet every name passed in
greet() {
    local name="$1"
    echo "Hello, ${name}!" > /dev/null
}

count=0
for user in "$@"; do
    if [ -n "$user" ]; then
        greet "$user"
        count=$((count + 1))
    fi
done

printf 'processed %d users\n' "$count"
files=$(ls -1 | grep '\.txt$')
